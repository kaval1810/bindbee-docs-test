## What Bindbee ships

| Package | Purpose | Install |
| --- | --- | --- |
| **`@bindbee/react-link`** | React hook for the Bindbee Embed connection flow | `npm install --save @bindbee/react-link` |
| **JavaScript CDN** | The same flow for any non-React frontend | `<script src="https://cdn.bindbee.dev/initialize.min.js">` |

<Info>
  These are **frontend** packages. They render the connection flow your customer
  uses to authorize their HR system — they are not clients for the unified data
  API.
</Info>

Everything else — reading employees, benefits, payroll, writing data back — is a plain REST API. Call it with your language's standard HTTP client. There is nothing to install.

<Note>
  If you need a server-side SDK for your language, tell
  `support@bindbee.dev` — demand shapes what gets built.
</Note>

## Frontend: React

```bash
npm install --save @bindbee/react-link
# or
yarn add @bindbee/react-link
```

```jsx
import { useBindbeeMagiclink } from "@bindbee/react-link";

function ConnectHrisButton({ linkToken, onConnected }) {
  const { open } = useBindbeeMagiclink({
    linkToken,                             // from your backend
    serverUrl: "https://api.bindbee.dev",  // 'https://api-eu.bindbee.dev' for EU
    onSuccess: (temporaryToken) => onConnected(temporaryToken),
    onClose: () => {},
  });

  return <button onClick={open}>Connect your HRIS</button>;
}
```

| Option | Required | Notes |
| --- | --- | --- |
| `linkToken` | Yes | Generated server-side; short-lived |
| `serverUrl` | No | Set explicitly for the EU region |
| `onSuccess` | Yes | Receives `temporary_token` — send it to your backend |
| `onClose` | No | Fires when the customer dismisses the flow |


```js
window.BindbeeEmbed.initialize({
  linkToken: LINK_TOKEN,
  serverUrl: "https://api.bindbee.dev",
  onSuccess: (temporaryToken) => sendToBackend(temporaryToken),
  onClose: () => {},
});

document.querySelector("#connect")
  .addEventListener("click", () => window.BindbeeEmbed.open());

// window.BindbeeEmbed.destroy() when unmounting
```

Keep the `integrity` and `crossorigin` attributes — they enable Subresource Integrity, so a compromised CDN can't inject code into your page.

<Warning>
  Only the `link_token` may reach the browser. Your API key and every
  `connector_token` are server-side secrets. The `temporary_token` →
  `connector_token` exchange must happen on your backend.
</Warning>

Full walkthrough: [Bindbee Embed](/sdk/bindbee-embed).

## Backend: a client worth copying

Two headers, cursor pagination, and retry on `429`/`5xx`. That is the whole client.

<CodeGroup>

```python Python
import os, time, requests

class BindbeeClient:
    RETRYABLE = {429, 500, 502, 503, 504}

    def __init__(self, api_key=None, base_url="https://api.bindbee.dev"):
        self.api_key = api_key or os.environ["BINDBEE_API_KEY"]
        self.base_url = base_url.rstrip("/")
        self.session = requests.Session()

    def _headers(self, connector_token=None):
        headers = {"Authorization": f"Bearer {self.api_key}"}
        if connector_token:
            headers["X-Connector-Token"] = connector_token
        return headers

    def request(self, method, path, connector_token=None, attempts=5, **kwargs):
        url = f"{self.base_url}{path}"
        for attempt in range(attempts):
            resp = self.session.request(
                method, url,
                headers=self._headers(connector_token),
                timeout=30, **kwargs,
            )
            if resp.status_code not in self.RETRYABLE:
                resp.raise_for_status()
                return resp

            delay = (int(resp.headers.get("Retry-After", 60))
                     if resp.status_code == 429
                     else min(2 ** attempt, 30))
            if attempt == attempts - 1:
                resp.raise_for_status()
            time.sleep(delay)

    def get(self, path, connector_token=None, **params):
        return self.request("GET", path, connector_token, params=params).json()

    def paginate(self, path, connector_token, **filters):
        cursor = None
        while True:
            params = {"page_size": 200, **filters}
            if cursor:
                params["cursor"] = cursor
            body = self.get(path, connector_token, **params)
            yield from body["items"]
            cursor = body.get("cursor")
            if not cursor:
                return


client = BindbeeClient()
for employee in client.paginate("/api/hris/v1/employees", token,
                                employment_status="ACTIVE"):
    upsert(employee)
```

```javascript Node / TypeScript
const RETRYABLE = new Set([429, 500, 502, 503, 504]);
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

class BindbeeClient {
  constructor({ apiKey = process.env.BINDBEE_API_KEY,
                baseUrl = "https://api.bindbee.dev" } = {}) {
    this.apiKey = apiKey;
    this.baseUrl = baseUrl.replace(/\/$/, "");
  }

  #headers(connectorToken) {
    const headers = { Authorization: `Bearer ${this.apiKey}` };
    if (connectorToken) headers["X-Connector-Token"] = connectorToken;
    return headers;
  }

  async request(path, { connectorToken, attempts = 5, ...init } = {}) {
    for (let attempt = 0; attempt < attempts; attempt++) {
      const res = await fetch(`${this.baseUrl}${path}`, {
        ...init,
        headers: { ...this.#headers(connectorToken), ...init.headers },
      });

      if (!RETRYABLE.has(res.status)) {
        if (!res.ok) throw new Error(`${res.status} ${await res.text()}`);
        return res;
      }

      const delay =
        res.status === 429
          ? Number(res.headers.get("Retry-After") ?? 60) * 1000
          : Math.min(2 ** attempt, 30) * 1000;

      if (attempt === attempts - 1) {
        throw new Error(`${res.status} after ${attempts} attempts`);
      }
      await sleep(delay);
    }
  }

  async get(path, { connectorToken, ...params } = {}) {
    const qs = new URLSearchParams(params);
    const res = await this.request(`${path}?${qs}`, { connectorToken });
    return res.json();
  }

  async *paginate(path, connectorToken, filters = {}) {
    let cursor = null;
    do {
      const body = await this.get(path, {
        connectorToken, page_size: 200, ...filters,
        ...(cursor ? { cursor } : {}),
      });
      yield* body.items;
      cursor = body.cursor;
    } while (cursor);
  }
}
```

```go Go
package bindbee

import (
	"encoding/json"
	"fmt"
	"net/http"
	"net/url"
	"os"
	"time"
)

type Client struct {
	APIKey  string
	BaseURL string
	HTTP    *http.Client
}

func New() *Client {
	return &Client{
		APIKey:  os.Getenv("BINDBEE_API_KEY"),
		BaseURL: "https://api.bindbee.dev",
		HTTP:    &http.Client{Timeout: 30 * time.Second},
	}
}

type Page struct {
	Cursor   *string           `json:"cursor"`
	PageSize int               `json:"page_size"`
	Items    []json.RawMessage `json:"items"`
}

func (c *Client) GetPage(path, connectorToken string, q url.Values) (*Page, error) {
	req, err := http.NewRequest("GET", c.BaseURL+path+"?"+q.Encode(), nil)
	if err != nil {
		return nil, err
	}
	req.Header.Set("Authorization", "Bearer "+c.APIKey)
	if connectorToken != "" {
		req.Header.Set("X-Connector-Token", connectorToken)
	}

	resp, err := c.HTTP.Do(req)
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()

	if resp.StatusCode != http.StatusOK {
		return nil, fmt.Errorf("bindbee: unexpected status %d", resp.StatusCode)
	}

	var page Page
	if err := json.NewDecoder(resp.Body).Decode(&page); err != nil {
		return nil, err
	}
	return &page, nil
}

// Paginate walks every page, calling fn for each item.
func (c *Client) Paginate(path, connectorToken string, q url.Values,
	fn func(json.RawMessage) error) error {

	if q == nil {
		q = url.Values{}
	}
	q.Set("page_size", "200")

	for {
		page, err := c.GetPage(path, connectorToken, q)
		if err != nil {
			return err
		}
		for _, item := range page.Items {
			if err := fn(item); err != nil {
				return err
			}
		}
		if page.Cursor == nil || *page.Cursor == "" {
			return nil
		}
		q.Set("cursor", *page.Cursor)
	}
}
```

<Tip>
  Make `connector_token` a **per-call** parameter, never a constructor argument.
  You will serve many customers through one client, and a client bound to one
  token is how cross-tenant data leaks get written.
</Tip>

## Configuration checklist

| Setting | Value |
| --- | --- |
| Base URL | `https://api.bindbee.dev` — or `https://api-eu.bindbee.dev` for EU |
| `Authorization` | `Bearer <api_key>` — prefix required |
| `X-Connector-Token` | On every customer-data request |
| Timeout | 30s is comfortable |
| Retry | `429` and `5xx` only, with backoff |
| `page_size` | 200 for bulk, lower with `include_raw_data` |
| TLS verification | Always on |

<Warning>
  Use the same region everywhere for a given customer — backend calls, the SDK's
  `serverUrl`, and the token exchange. A connector created in one region is
  invisible from the other.
</Warning>
